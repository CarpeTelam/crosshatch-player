# Game API level 1 (preview): everything a level-1 host offers a game.
#
# This list owns the level (spine AD-19). A host's surface is the union of
# api-level-<API_MIN_LEVEL>.txt to api-level-<API_LEVEL>.txt, and
# lib/GameCore/ApiLevel.h holds API_SURFACE_CRC over those files. Level 1 is a
# preview until API_LEVEL_FROZEN is set: it may still grow. Once a level is
# frozen, its file never changes; additions open the next level.
#
# Grammar. Lines starting with '#' and blank lines are ignored. Every other line
# is one entry, ASCII, words separated by single spaces, no trailing space:
#
#   <kind> <body>
#
#   fn <path>(<params>)[ -> <returns>]
#       A host function. <path> is a dotted Lua name; a path without a dot is a
#       global. <params> are names separated by ", "; a trailing '?' marks an
#       optional one, "..." any number of values.
#   field <path> <type>
#       A value under ch that is not a function.
#   enum <set> <value>
#       One accepted string of an argument or field set.
#   event <kind>[ <field>...]
#       One event table passed to input(); every event also has `kind`.
#   ctx <name> <type>
#       One field of the ctx table passed to setup().
#   manifest <key> <type>
#       One manifest.json key; nested keys are dotted, a trailing '?' on the type
#       marks an optional key.
#   limit <name> <integer>
#       One named limit; the unit ends the name (_bytes, _ms, _count, ...).
#   lib <global>[.<member>]
#       One global from Lua 5.5.1's standard libraries (AD-4), or one member of
#       a library table; every member a game can reach is listed after its table.
#   icon <name>
#       One library icon name (appended by epic-icon-library): Phosphor's own,
#       [a-z][a-z0-9-]{0,31}, each drawn in every `enum weight` value.
#   draw <fn path> ink|opaque
#       How a ch.gfx function's pixels reach the canvas (appended by e3r-1, the
#       icon-library retrospective's follow-up): ink draws only the shape's own
#       pixels in the color and leaves the rest as it was; opaque draws every
#       pixel of its rectangle.
#   name <what> <pattern>
#       The names a game gives something of its own (appended by e3r-1, the
#       icon-library retrospective's follow-up): <what> is lower-case letters
#       and '_'; every name must match <pattern>, an ECMAScript regular
#       expression with no spaces, matched whole.
#   seats_max <integer>
#       The most seats a match can have.
#
# The script's globals are ch and the fn and lib entries without a dot.
#
# API_SURFACE_CRC is the CRC-32 (zlib's) of every entry line of the level files
# in level order, each followed by "\n", leaving out comments and blank lines;
# test/game_core/ApiLevelTest.cpp recomputes it. Change the define with the list.
#
# Contract and comments. Only entry lines are contract: they alone feed
# API_SURFACE_CRC, which preview devices are matched by (AD-13). Behaviour a
# comment describes is documentation, outside the CRC and the preview match: a
# preview can change it without changing the CRC. Comments are still frozen
# with the rest of the file once the level freezes (the API freeze job compares
# the whole file). A rule a game may rely on across hosts becomes an entry (a
# limit, an enum, a signature) before its level freezes.

# Coordinates and sizes (x, y, w, h, r, x1, y1, x2, y2) saturate to
# -32768..32767 when the command is recorded, before it is clipped to the
# canvas. So rect(-40000, 0, 80480, 800, c, true) is recorded as

# How ch.gfx.icon and ch.gfx.image reach the canvas. An icon draws only its ink
# pixels, in its color, so what is under the rest of its square shows through. An
# image draws every pixel of its rectangle: with "black" its black pixels black and
# its white pixels white, as converted; with "white" the two swapped.

# the value the player chose on the game's Options screen, as a string; every

# icon is a library icon name: it matches the pattern below whole ("_" is not
# accepted) and takes at most manifest_icon_bytes bytes, and a name the library
# lacks makes the package invalid (the installer and scripts/pack_game.py check
# it; Manifest::parse cannot see the library). icon_weight is "regular" (when absent) or
# "fill", ch.gfx.icon's weights (the enum icon_weight entries below); any other
# value makes the package invalid. What a game without an icon shows is up to the
# launcher.
name manifest_icon [a-z][a-z0-9]*(-[a-z0-9]+)*
enum icon_weight regular
enum icon_weight fill
# default_mode is one of the manifest's modes (an enum mode value): the mode New
# game starts until the player picks another in Options. Absent, it is solo when
# modes lists solo, else the first this host can start in solo, pass, nearby
# order; a value not in modes makes the package invalid, and one this host
# cannot start falls back the same way. The player's last started mode wins over
# it once there is one.
# settings is an array of at most settings_count objects, each with exactly the
# keys id, name, values, and default (any other key makes the package invalid):
# id matches the setting_id pattern below and is unique; name is 1 to
# setting_name_bytes bytes; values holds 2 to setting_values_count unique
# strings of 1 to setting_value_bytes bytes; default is one of values (absent:
# the first). A remembered choice the manifest no longer declares falls back to
# that default.
name setting_id [a-z][a-z0-9_]{0,15}

# The canvas pixels one frame's ch.gfx.icon and ch.gfx.image calls may cover,
# summed over the calls. Each counts the part of its square (an icon's drawn
# size) or rectangle (an image's own size) inside the canvas, at its coordinates
# as recorded (saturated as above), counting only pixels inside this host's
# canvas (ch.screen.w x ch.screen.h), so a call wholly off the canvas counts
# nothing, and overlapping calls count each time. About 2.9 canvases on the X4
# Pro (466 x 788 is 367,208) and 2.8 on the Sticky (474 x 788 is 373,512): the host replays a frame by walking
# each covered pixel, so this bounds that icon and image replay only; filled
# rects and circles are not charged. The call that would pass it stops the game
# with "ch.gfx.<icon or image>: the frame's icons and images cover over 1048576
# pixels", pcall or not, and that frame is not shown. Each frame starts again
# from 0.
limit frame_icon_image_pixels 1048576
# ch.gfx.icon's drawn sizes, small, medium, and large, in pixels a side: large is
# the medium bitmap with each pixel drawn as a 2 x 2 block.

limit icon_medium_side_pixels 64
limit icon_large_side_pixels 128

# A game's own images (ch.gfx.image): the .bmp files in its folder, their bytes
# with headers included and icon.bmp excluded, and how many there are. A w x h
# image's .bmp takes 62 + ceil(w / 32) * 4 * h bytes.
limit images_bytes 131072
limit images_count 32
# A .chgame package (docs/crosshatch/formats.md): the file's bytes, its members (a folder is
# not allowed), one member's bytes uncompressed, the .lua members' bytes together, and the stem
# of a .lua or .png member's name ([a-z0-9_]{1,32}). The installer rejects a package over any
# of them; scripts/pack_game.py refuses to make one over any of them. Also the deepest nesting of
# manifest.json's own objects and arrays (the device's JSON parser, StreamingJsonParser::MAX_NESTING,
# refuses a 33rd open bracket; the packer refuses the same manifest).
limit package_bytes 262144
limit package_members_count 64
limit member_bytes 131072
limit lua_sources_bytes 262144
limit member_stem_bytes 32
limit manifest_nesting_count 32
# A .png member's largest width and height in pixels (icon.png is scaled to 64 x 64).
limit image_width_pixels 2048
limit image_height_pixels 3072
# Two more reserved images, which only the runtime draws: title.png, the title
# screen's splash, drawn centred and clipped in a 480 x 480 area under its header
# (absent or unreadable: the game's icon at 128 px), and handoff.png, the hidden
# pass hand-off screen's splash, drawn centred and clipped in the same 480 x 480
# area, above the runtime's "Player N's turn" and "I'm ready" button (absent or
# unreadable: title.png, else the game's icon at 128 px). One larger than its
# limits below makes the package invalid. Both count toward images_bytes and
# images_count; icon.png does not.
limit title_image_width_pixels 480
limit title_image_height_pixels 480
limit handoff_image_width_pixels 480
limit handoff_image_height_pixels 480
# An image's name, which ch.gfx.image takes, is its file name without ".bmp".
# Unlike icon names, it takes no "-": a file named otherwise is skipped with a log
# line, and drawing it stops the game as an unknown image. icon.bmp is the
# launcher's, and title.bmp and handoff.bmp are the runtime's pages: none is ever
# an image, so the pattern leaves out "icon", "title", and "handoff", and drawing
# one of them stops the game as an unknown image.
name image (?!(icon|title|handoff)$)[a-z0-9_]{1,32}
# Lua's C-level limit (LUAI_MAXCCALLS in lib/lua/library.json): nested C calls
# (pcall, xpcall, metamethods, string.gsub and table.sort callbacks, require) and
# the parser's nesting of expressions and blocks (right-associative .. or ^
# chains, nested parentheses, tables, functions, and blocks, about 28 deep)
# share it (a pcall takes one level, an xpcall two: the runtime's wrapper and
# Lua's own, so about 28 nested pcalls or 14 nested xpcalls fit); past it the
# call or the load fails with "C stack overflow", an ordinary Lua error that
# pcall can catch. The runtime's own C stack headroom check ("script recursion
# too deep") is the one that stops the game.
limit c_stack_levels_count 30
# Lua's pattern-matcher depth (MAXCCALLS in lib/lua/library.json): a match nests
# one level, plus one for each capture's start and end and each item with ?, *,
# +, or - before the rest of the pattern ("(%w+)=(%w+)" takes 7); past it
# string.find, match, gmatch, and gsub fail with "pattern too complex".
limit pattern_depth_count 16
