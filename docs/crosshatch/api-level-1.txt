# Game API level 1 (preview): everything a level-1 host offers a game.
#
# This list owns the level (spine AD-19). A host's surface is the union of
# api-level-<API_MIN_LEVEL>.txt to api-level-<API_LEVEL>.txt, and
# lib/GameCore/ApiLevel.h holds API_SURFACE_CRC over those files. Level 1 is a
# preview until API_LEVEL_FROZEN is set: it may still grow. Once a level is
# frozen, its file never changes; additions open the next level.
#
# Grammar. Lines starting with '#' and blank lines are ignored. Every other line
# is one entry, ASCII, words separated by single spaces, no trailing space:
#
#   <kind> <body>
#
#   fn <path>(<params>)[ -> <returns>]
#       A host function. <path> is a dotted Lua name; a path without a dot is a
#       global. <params> are names separated by ", "; a trailing '?' marks an
#       optional one, "..." any number of values.
#   field <path> <type>
#       A value under ch that is not a function.
#   enum <set> <value>
#       One accepted string of an argument or field set.
#   event <kind>[ <field>...]
#       One event table passed to input(); every event also has `kind`.
#   ctx <name> <type>
#       One field of the ctx table passed to setup().
#   manifest <key> <type>
#       One manifest.json key; nested keys are dotted, a trailing '?' on the type
#       marks an optional key.
#   limit <name> <integer>
#       One named limit; the unit ends the name (_bytes, _ms, _count, ...).
#   lib <global>[.<member>]
#       One global from Lua 5.5.1's standard libraries (AD-4), or one member of
#       a library table; every member a game can reach is listed after its table.
#   icon <name>
#       One library icon name (appended by epic-icon-library): Phosphor's own,
#       [a-z][a-z0-9-]{0,31}, each drawn in every `enum weight` value.
#   seats_max <integer>
#       The most seats a match can have.
#
# The script's globals are ch and the fn and lib entries without a dot.
#
# API_SURFACE_CRC is the CRC-32 (zlib's) of every entry line of the level files
# in level order, each followed by "\n", leaving out comments and blank lines;
# test/game_core/ApiLevelTest.cpp recomputes it. Change the define with the list.
#
# Contract and comments. Only entry lines are contract: they alone feed
# API_SURFACE_CRC, which preview devices are matched by (AD-13). Behaviour a
# comment describes is documentation, outside the CRC and the preview match: a
# preview can change it without changing the CRC. Comments are still frozen
# with the rest of the file once the level freezes (the API freeze job compares
# the whole file). A rule a game may rely on across hosts becomes an entry (a
# limit, an enum, a signature) before its level freezes.

# Coordinates and sizes (x, y, w, h, r, x1, y1, x2, y2) saturate to
# -32768..32767 when the command is recorded, before it is clipped to the
# canvas. So rect(-40000, 0, 80480, 800, c, true) is recorded as
# rect(-32768, 0, 32767, 800, c, true), which covers x = -32768..-2 and draws
# nothing, not the full-width rect the wide values describe; a circle or text
# anchored beyond the range moves to its edge.

# Globals the runtime provides: print writes to ch.log; require loads name.lua
# from the game's own files; setmetatable is Lua's, except that it refuses a
# metatable with a __gc field (finalizers would run outside the budget) or a
# __close field. A __close added to a metatable later still runs, but cannot turn
# the heap cap into an error pcall catches: the runtime records it where Lua
# raises it.

# A game's own images (ch.gfx.image): the .bmp files in its folder, their bytes
# with headers included and icon.bmp excluded, and how many there are. A w x h
# image's .bmp takes 62 + ceil(w / 32) * 4 * h bytes.
# An image's name is its file name without ".bmp": 1 to 32 characters from [a-z0-9_]. Unlike icon names, it takes no
# "-"; a file named otherwise is skipped with a log line, and drawing it stops the game as an unknown image.
limit images_bytes 131072
limit images_count 32
# Lua's C-level limit (LUAI_MAXCCALLS in lib/lua/library.json): nested C calls
# (pcall, xpcall, metamethods, string.gsub and table.sort callbacks, require) and
# the parser's nesting of expressions and blocks (right-associative .. or ^
# chains, nested parentheses, tables, functions, and blocks, about 28 deep)
# share it (a pcall takes one level, an xpcall two: the runtime's wrapper and
# Lua's own, so about 28 nested pcalls or 14 nested xpcalls fit); past it the
# call or the load fails with "C stack overflow", an ordinary Lua error that
# pcall can catch. The runtime's own C stack headroom check ("script recursion
# too deep") is the one that stops the game.
limit c_stack_levels_count 30
# Lua's pattern-matcher depth (MAXCCALLS in lib/lua/library.json): a match nests
# one level, plus one for each capture's start and end and each item with ?, *,
# +, or - before the rest of the pattern ("(%w+)=(%w+)" takes 7); past it
# string.find, match, gmatch, and gsub fail with "pattern too complex".
limit pattern_depth_count 16
